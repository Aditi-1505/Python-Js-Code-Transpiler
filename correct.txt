from parser import Parser, FunctionDef, ClassDef, Import, FromImport, Program
def _has_runnable_statements(source_code):
    try:
        tokens = Lexer(source_code).tokenize()
        ast    = Parser(tokens).parse()
        for stmt in ast.statements:
            if not isinstance(stmt, (FunctionDef, ClassDef, Import, FromImport)):
                return True
        return False
    except Exception:
        return True

def add(a, b):
    return a + b
def subtract(a, b):
    return a - b
def multiply(a, b):
    return a * b
def divide(a, b):
    if b == 0:
        return "Cannot divide by zero"
    return a / b
def main():
    print("Enter two numbers:")
    x = float(input("First number: "))
    y = float(input("Second number: "))
    print("\nChoose operation:")
    print("1. Add")
    print("2. Subtract")
    print("3. Multiply")
    print("4. Divide")
    choice = input("Enter choice (1-4): ")
    if choice == '1':
        print("Result:", add(x, y))
    elif choice == '2':
        print("Result:", subtract(x, y))
    elif choice == '3':
        print("Result:", multiply(x, y))
    elif choice == '4':
        print("Result:", divide(x, y))
    else:
        print("Invalid choice")
main()

# Function to take list input
def create_list():
    n = int(input("Enter number of elements: "))
    lst = []
    for i in range(n):
        val = int(input(f"Enter element {i+1}: "))
        lst.append(val)
    return lst

# Function to find sum of list
def find_sum(lst):
    return sum(lst)

# Function to find maximum element
def find_max(lst):
    return max(lst)

# Function to find minimum element
def find_min(lst):
    return min(lst)

# Function to sort list
def sort_list(lst):
    return sorted(lst)

# Function to display list
def display(lst):
    print("List:", lst)

# Main function
def main():
    my_list = create_list()

    display(my_list)
    print("Sum:", find_sum(my_list))
    print("Maximum:", find_max(my_list))
    print("Minimum:", find_min(my_list))
    print("Sorted List:", sort_list(my_list))

main()
